# Face, hand and emotion detection.
# mediapipe 0.10.30 and newer installs on any Python 3 version, including 3.13
# and 3.14. Intel Macs only have builds up to 0.10.21, which needs Python 3.12
# or older there. The app uses the API both versions share.
opencv-contrib-python>=4.8,<5
mediapipe>=0.10.21,<0.11
numpy>=1.23

# Runs the bundled emotion model. ai-edge-litert is about 15MB and covers
# Apple Silicon Macs, Windows and Linux. Python 3.9 and Intel Macs have no
# wheel for it, so they use tensorflow, which is much larger but works.
ai-edge-litert>=2.0,<3; python_version >= "3.10" and (sys_platform != "darwin" or platform_machine != "x86_64")
tensorflow>=2.13,<3; python_version < "3.10" or (sys_platform == "darwin" and platform_machine == "x86_64")

# Draws the reaction captions with the bundled font, and plays animated reactions.
Pillow>=10,<13

# Fetches real cat and dog reaction media, and talks to the optional OpenAI API for live captions.
requests>=2.28,<3
openai>=1.0,<2
